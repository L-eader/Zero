كود الجسر الكامل (server.js) لربط RCON + الشات العربي + Pterodactyl API

انسخ هذا الكود وشغّله على Node.js أو استضافة Render ليربط لوحة تحكم موقعك بالسيرفر والاستضافة مباشرة

التثبيت: npm install express cors rcon-client axios 
const express = require('express');
const cors = require('cors');
const axios = require('axios');
const { Rcon } = require('rcon-client');

const app = express();
app.use(cors());
app.use(express.json());

// 1. إرسال رسائل الشات العربي أو أوامر الأدمن عبر RCON
app.post('/api/rcon', async (req, res) => {
    const { host, port, password, command } = req.body;
    try {
        const rcon = await Rcon.connect({ host, port: parseInt(port), password });
        const response = await rcon.send(command);
        await rcon.end();
        res.json({ success: true, response });
    } catch (err) {
        res.status(500).json({ success: false, error: err.message });
    }
});

// 2. التحكم بالاستضافة (تشغيل / إيقاف / ريستارت) عبر Pterodactyl API Key
app.post('/api/power', async (req, res) => {
    const { panelUrl, serverId, apiKey, signal } = req.body; // signal: start | stop | restart | kill
    try {
        await axios.post(`${panelUrl}/api/client/servers/${serverId}/power`, 
            { signal },
            { headers: { 'Authorization': `Bearer ${apiKey}`, 'Accept': 'application/json' } }
        );
        res.json({ success: true, message: `Power signal ${signal} sent!` });
    } catch (err) {
        res.status(500).json({ success: false, error: err.message });
    }
});

app.listen(3000, () => console.log('ZeroSMP Full Admin & Chat Bridge running on port 3000'));
