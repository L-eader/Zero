كود الجسر الحقيقي (server.js) لربط RCON والبانل

شغّل هذا الكود بجانب بوت الديسكورد الخاص بك (`npm i express cors rcon-client axios`) ثم ضع رابطه في `SERVER_CONFIG.rcon.bridgeUrl` أعلى كود الموقع ليعمل الـ RCON والتحكم بالبانل وجلب الرام والتخزين الحقيقي بدون حظر CORS: 

const express = require('express');
const cors = require('cors');
const { Rcon } = require('rcon-client');
const axios = require('axios');

const app = express();
app.use(cors());
app.use(express.json());

// 1. إرسال أوامر RCON والرسائل العربية إلى سيرفر ماينكرافت
app.post('/api/rcon', async (req, res) => {
  const { host, port, password, command } = req.body;
  try {
    const rcon = await Rcon.connect({ host, port: Number(port), password });
    const response = await rcon.send(command);
    await rcon.end();
    res.json({ success: true, response });
  } catch (err) {
    res.status(500).json({ success: false, error: err.message });
  }
});

// 2. جلب استهلاك الرام والتخزين والمعالج الحقيقي من Pterodactyl
app.post('/api/resources', async (req, res) => {
  const { panelUrl, serverId, apiKey } = req.body;
  try {
    const base = panelUrl.replace(/\/$/, '');
    const headers = { 'Authorization': `Bearer ${apiKey}`, 'Accept': 'Application/vnd.pterodactyl.v1+json' };
    const [resUsage, resDetails] = await Promise.all([
      axios.get(`${base}/api/client/servers/${serverId}/resources`, { headers }),
      axios.get(`${base}/api/client/servers/${serverId}`, { headers })
    ]);
    res.json({
      success: true,
      resources: resUsage.data.attributes,
      limits: resDetails.data.attributes.limits
    });
  } catch (err) {
    res.status(500).json({ success: false, error: err.message });
  }
});

// 3. إرسال أوامر الطاقة (start / stop / restart / kill) للبانل
app.post('/api/power', async (req, res) => {
  const { panelUrl, serverId, apiKey, signal } = req.body;
  try {
    const base = panelUrl.replace(/\/$/, '');
    await axios.post(`${base}/api/client/servers/${serverId}/power`, { signal }, {
      headers: { 'Authorization': `Bearer ${apiKey}`, 'Accept': 'Application/vnd.pterodactyl.v1+json' }
    });
    res.json({ success: true });
  } catch (err) {
    res.status(500).json({ success: false, error: err.message });
  }
});

app.listen(process.env.PORT || 3000, () => console.log('ZeroSMP Bridge Ready'));
